cmake_minimum_required(VERSION 3.20)
project(DeltaTextMsdf LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MSDFGEN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/msdfgen")
file(GLOB MSDFGEN_CORE_SOURCES CONFIGURE_DEPENDS "${MSDFGEN_ROOT}/core/*.cpp")
add_library(DeltaTextMsdf SHARED DeltaTextMsdf.cpp ${MSDFGEN_CORE_SOURCES})
target_include_directories(DeltaTextMsdf PRIVATE "${MSDFGEN_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(DeltaTextMsdf PRIVATE MSDFGEN_USE_CPP11 MSDFGEN_VERSION=1.13 MSDFGEN_VERSION_MAJOR=1 MSDFGEN_VERSION_MINOR=13 MSDFGEN_VERSION_REVISION=0)
set_target_properties(DeltaTextMsdf PROPERTIES OUTPUT_NAME "DeltaTextMsdf")
